// Heterogeneous Vulkan Compute GEMM Kernel for Workstation Dual RX 9070
layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;
layout(push_constant) uniform PushConsts {
  uint M; // Matrix Rows
  uint N; // Matrix Columns
  uint K; // Inner Dimension
  float alpha;
} consts;
layout(std430, set = 0, binding = 0) readonly buffer SMatrixA { float dataA[]; };
layout(std430, set = 0, binding = 1) readonly buffer SMatrixB { float dataB[]; };
layout(std430, set = 0, binding = 2) writeonly buffer SMatrixC { float dataC[]; };
shared float tileA[16][16];
shared float tileB[16][16];
void main() {
  uint globalRow = gl_GlobalInvocationID.y;
  uint globalCol = gl_GlobalInvocationID.x;
  float acc = 0.0;
  uint numTiles = (consts.K + 15) / 16;
  for (uint t = 0; t < numTiles; ++t) {
    tileA[gl_LocalInvocationID.y][gl_LocalInvocationID.x] = dataA[globalRow * consts.K + (t * 16 + gl_LocalInvocationID.x)];
    tileB[gl_LocalInvocationID.y][gl_LocalInvocationID.x] = dataB[(t * 16 + gl_LocalInvocationID.y) * consts.N + globalCol];
    memoryBarrierShared(); barrier();
    for (int k = 0; k < 16; ++k) acc += tileA[gl_LocalInvocationID.y][k] * tileB[k][gl_LocalInvocationID.x];
    memoryBarrierShared(); barrier();
  }
  dataC[globalRow * consts.N + globalCol] = acc * consts.alpha;
}
